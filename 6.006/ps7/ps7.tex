\problem  % Problem 1
\\
\textbf{Subproblems:} $x(i)$: max delegates Torr can win in states $i$ to $n$, where campaigning on day $i$ is possible, for $i \in \{1, \dots, n+1\}$.

\textbf{Relate:} $x(i) = \max\{\, d_i + z_{i+1} + z_{i+2} + x(i+3),\ \ z_i + x(i+1) \,\}$

\textbf{Topological Order:} $n+1, n, \dots, 2, 1$ ($x(i)$ only depends on subproblems with strictly larger $i$)

\textbf{Base Cases:} $x(n+1) = 0, x(n) = d_n, x(n-1) = \max\{\,d_{n-1}+z_n,\ z_{n-1}+d_n\,\}$

\textbf{Original Problem:} $x(1) \geq \lfloor D/2 \rfloor + 1$

\textbf{Time:} $T(n) = T(n - 1) + O(1) \implies O(n)$


\problem  % Problem 2
\\
\textbf{Subproblems:} $x(i,j)$: min total discomfort matching tigers $T[i:]$ to a subsequence of $C[j:]$, for $i \in \{0,\dots,n\}$, $j \in \{0,\dots,n^2\}$

\textbf{Relate:} $x(i,j) = \min\{\, d(i,j) + x(i+1,j+1),\ \ x(i,j+1) \,\}$, where $d(i,j) = \max\{0,\ s_i - c_j\}$

\textbf{Topological Order:} $j = n^2, n^2-1, \dots, 1, 0$ ($x(i,j)$ only depends on subproblems with strictly larger $j$)

\textbf{Base Cases:} $x(n,j) = 0,\ \ x(i,n^2) = \infty$ for $i > 0$

\textbf{Original Problem:} $x(0,0)$

\textbf{Time:} : (n+1)(n²+1) subproblems $\times\ O(1) \implies O(n^3)$

\problem  % Problem 3: Odd Paths
\\
\textbf{Subproblems:} $x(v,i)$: number of paths from $s$ to $v$ with even weight if $i=0$, odd weight if $i=1$, for $v \in V$, $i \in \{0,1\}$.

\textbf{Relate:} $x(v,i) = \displaystyle\sum_{u \,\in\, \text{Adj}^-(v)} x\big(u,\ p(w(u,v)+i)\big)$, where $p(k)$ is the parity of $k$

\textbf{Topological Order:} topological order of $G$ ($x(v,i)$ only depends on subproblems for vertices earlier in the order)

\textbf{Base Cases:} $x(s,0) = 1,\ \ x(s,1) = 0,\ \ x(v,0) = x(v,1) = 0$ for $v$ with $\text{Adj}^-(v) = \emptyset$

\textbf{Original Problem:} $x(t,1)$

\textbf{Time:} $\displaystyle\sum_{v \in V} O(\deg^-(v)) = O(|V| + |E|)$


\problem  % Problem 4: Pizza Partitioning
\\
\textbf{Subproblems:} $x(i,j,p)$: max tastiness Liza can achieve with $j$ slices counter-clockwise from $\alpha_i$ remaining, Liza choosing if $p=1$, Lie choosing if $p=2$, for $i \in \{0,\dots,2n-1\}$, $j \in \{0,\dots,n\}$, $p \in \{1,2\}$

\textbf{Relate:}
\[
x(i,j,1) = \max_{k \,\in\, \{1,\dots,j-1\}} \big\{\, v(i,k) + x(i+k,\ j-k,\ 2),\ \ v(i+k,\ j-k) + x(i,\ k,\ 2) \,\big\}
\]
\[
x(i,j,2) = \min_{k \,\in\, \{1,\dots,j-1\}} \big\{\, x(i+k,\ j-k,\ 1),\ \ x(i,\ k,\ 1) \,\big\}
\]
where $v(i,j) = \sum_{k=0}^{j-1} t_{(i+k) \bmod 2n}$ (precomputed for all $i,j$ in $O(n^3)$ time)

\textbf{Topological Order:} $j = 1, 2, \dots, n$ ($x(i,j,p)$ only depends on subproblems with strictly smaller $j$)

\textbf{Base Cases:} $x(i,1,1) = t_i,\ \ x(i,1,2) = 0$

\textbf{Original Problem:} $\displaystyle\max_{i \,\in\, \{0,\dots,2n-1\}} \big\{\, x(i,n,2) + v\big((i+n) \bmod 2n,\ n\big) \,\big\}$

\textbf{Time:} $O(n^3)$ to compute all $v(i,j)$, plus $O(n^2)$ subproblems $\times\ O(n) \implies O(n^3)$


\problem  % Problem 5: Shorting Stocks
\\
\textbf{(a)}
\\
\textbf{Subproblems:} $x(j)$: length of longest decreasing no-skip-days subsequence in $P[j:]$ that includes $P[j]$, for $j \in \{0,\dots,nk-1\}$

\textbf{Relate:} $x(j) = 1 + \max\big(\{x(d) \mid d \in \{j+1,\dots,f(j)\},\ P[j] > P[d]\} \cup \{0\}\big)$, where $f(j) = \min\{j + (k-1) - (j \bmod k) + k,\ nk-1\}$

\textbf{Topological Order:} $j = nk-1, nk-2, \dots, 1, 0$ ($x(j)$ only depends on subproblems with strictly larger $j$)

\textbf{Base Cases:} $x(nk-1) = 1$ (also implied by relation when the max is over the empty set)

\textbf{Original Problem:} $\displaystyle\max_{j=0}^{nk-1} x(j)$, over all companies; store parent pointers to reconstruct $S$

\textbf{Time:} $O(nk)$ subproblems $\times\ O(k) \implies O(nk^2)$ per company, $O(snk^2)$ total
