  \item \textbf{build(X)}: Store all pages in a static array and initialize sections P1, P2, P3 with gaps in O(n).
  \item \textbf{place mark(i, m)}: Rebuild array S to insert bookmark m at index i, maintaining gaps between P1, P2, P3 in O(n).
  \item \textbf{read page(i)}: Compute which section (P1, P2, P3) contains index i and return page in O(1).

\problem  % Problem 4

\begin{problemparts}
\problempart % Problem 4a
\begin{itemize}
    \item \textbf{insert first(x):} Insert $x$ at the beginning by setting $x.next$ to the current head, updating the current head's previous pointer to $x$ if it exists, and then updating $L.head$ to $x$; if the list was empty, also set $L.tail$ to $x$.
    \item \textbf{insert last(x):} Insert $x$ at the end by setting $x.prev$ to the current tail, updating the current tail's next pointer to $x$ if it exists, and then updating $L.tail$ to $x$; if the list was empty, also set $L.head$ to $x$.
    \item \textbf{delete first():} Remove the first node by updating $L.head$ to $L.head.next$ and setting the new head's previous pointer to \text{None} if it exists; if the list becomes empty, set $L.tail$ to \text{None}.
    \item \textbf{delete last():} Remove the last node by updating $L.tail$ to $L.tail.prev$ and setting the new tail's next pointer to \text{None} if it exists; if the list becomes empty, set $L.head$ to \text{None}.
\end{itemize}
\problempart % Problem 4b
\textbf{Remove sublist from $x_1$ to $x_2$:} Disconnect the sublist by setting $x_1.prev.next$ to $x_2.next$ (if $x_1$ is not the head) and $x_2.next.prev$ to $x_1.prev$ (if $x_2$ is not the tail). Update $L.head$ to $x_2.next$ \textbf{if} $x_1$ was the head, and $L.tail$ to $x_1.prev$ \textbf{if} $x_2$ was the tail. Set $x_1.prev$ to \text{None} and $x_2.next$ to \text{None} to isolate the sublist, then return it as a new doubly linked list with head $x_1$ and tail $x_2$.
\problempart % Problem 4c
\textbf{Splice $L_2$ into $L_1$ after node $x$:} If $L_2$ is not empty, set $L_2.head.prev = x$ and $L_2.tail.next = x.next$; if $x.next$ exists, set $x.next.prev = L_2.tail$; then set $x.next = L_2.head$. If $x$ was the tail of $L_1$, update $L_1.tail = L_2.tail$. Finally, set $L_2.head = \text{None}$ and $L_2.tail = \text{None}$ to empty $L_2$.
\end{problemparts}
